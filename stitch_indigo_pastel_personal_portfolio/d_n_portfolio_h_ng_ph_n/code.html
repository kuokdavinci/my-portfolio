// HNSW Insertion Path
fn insert(&mut self, vec: &[f32], id: u64) {
    let mut ep = self.enter_point;
    let max_layer = self.random_layer();
    for lc in (0..self.max_level).rev() {
        ep = self.search_layer(vec, ep, 1, lc);
        if lc <= max_layer {
            self.connect_node(id, ep, lc);
        }
    }
}
