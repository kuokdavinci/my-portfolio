Create `src/modules/dashboard/dashboard.js` — the main dashboard orchestrator that assembles all 5 sections.

```javascript
import { renderSystemOverview } from './sections/system-overview.js';
import { renderUserBehavior } from './sections/user-behavior.js';
import { renderAIObservability } from './sections/ai-observability.js';
import { renderInfrastructureHealth } from './sections/infrastructure-health.js';
import { renderLiveTelemetry } from './sections/live-telemetry.js';

let refreshInterval = null;
let sseConnection = null;

export async function initDashboard() {
  const container = document.getElementById('dashboard-content');
  if (!container) return;

  // Clear existing content
  container.innerHTML = '';

  // Render all 5 sections
  await renderSystemOverview(createSectionContainer(container, 'system-overview'));
  await renderUserBehavior(createSectionContainer(container, 'user-behavior'));
  await renderAIObservability(createSectionContainer(container, 'ai-observability'));
  await renderInfrastructureHealth(createSectionContainer(container, 'infrastructure-health'));
  sseConnection = renderLiveTelemetry(createSectionContainer(container, 'live-telemetry'));

  // Set up unified 10s refresh cycle
  refreshInterval = setInterval(async () => {
    await refreshDashboard();
  }, 10000);
}

export async function refreshDashboard() {
  // Re-render all sections with fresh data
  // Each section's render function should handle its own data fetching
  const overviewContainer = document.getElementById('section-system-overview');
  const behaviorContainer = document.getElementById('section-user-behavior');
  const aiContainer = document.getElementById('section-ai-observability');
  const infraContainer = document.getElementById('section-infrastructure-health');

  if (overviewContainer) await renderSystemOverview(overviewContainer, { refresh: true });
  if (behaviorContainer) await renderUserBehavior(behaviorContainer, { refresh: true });
  if (aiContainer) await renderAIObservability(aiContainer, { refresh: true });
  if (infraContainer) await renderInfrastructureHealth(infraContainer, { refresh: true });
  // Live telemetry updates via SSE, no need to re-render
}

export function destroyDashboard() {
  if (refreshInterval) {
    clearInterval(refreshInterval);
    refreshInterval = null;
  }
  if (sseConnection) {
    sseConnection.close();
    sseConnection = null;
  }
  // Destroy all Chart.js instances
  if (window.myCharts) {
    Object.values(window.myCharts).forEach(chart => chart.destroy());
    window.myCharts = {};
  }
}

function createSectionContainer(parent, id) {
  const section = document.createElement('div');
  section.id = `section-${id}`;
  section.className = 'dashboard-section';
  parent.appendChild(section);
  return section;
}
```
