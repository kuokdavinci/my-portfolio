```javascript
// src/modules/tracking/tracker.js
const TRACKING_API_URL = 'http://localhost:8000/api/v1/track';
const HEARTBEAT_URL = 'http://localhost:8000/api/v1/heartbeat';

class Tracker {
  constructor() {
    this.queue = [];
    this.flushInterval = null;
    this.heartbeatInterval = null;
    this.retryQueue = [];
    this.sessionId = null;
    this.sessionStart = null;
  }

  getSessionId() {
    if (!this.sessionId) {
      this.sessionId = sessionStorage.getItem('portfolio_session_id');
      if (!this.sessionId) {
        this.sessionId = 'session_' + Math.random().toString(36).substring(2, 15) + Math.random().toString(36).substring(2, 15);
        sessionStorage.setItem('portfolio_session_id', this.sessionId);
      }
    }
    return this.sessionId;
  }

  async init() {
    this.getSessionId();
    this.sessionStart = Date.now();
    this.track('session_start', { session_id: this.sessionId });
    this.flushInterval = setInterval(() => this.flush(), 5000);
    this.startHeartbeat();
    this.bindScrollTracking();
    this.bindUnloadHandler();
  }

  track(eventType, payload = {}) {
    this.queue.push({
      session_id: this.getSessionId(),
      timestamp: new Date().toISOString(),
      event_type: eventType,
      payload: payload
    });
    if (this.queue.length >= 10) {
      this.flush();
    }
  }

  async flush() {
    if (this.queue.length === 0) return;
    const batch = [...this.queue];
    this.queue = [];
    try {
      await Promise.all(batch.map(event => this.sendEvent(event)));
    } catch (err) {
      this.retryQueue.push(...batch);
      this.scheduleRetry();
    }
  }

  async sendEvent(event) {
    try {
      const response = await fetch(TRACKING_API_URL, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(event)
      });
      if (!response.ok) throw new Error(response.statusText);
    } catch (err) {
      // Beacon API fallback for critical events
      if (event.event_type === 'session_end') {
        navigator.sendBeacon(TRACKING_API_URL, JSON.stringify(event));
      }
      throw err;
    }
  }

  scheduleRetry() {
    // Retry up to 3 times with exponential backoff
    if (this.retryQueue.length === 0) return;
    setTimeout(() => {
      const batch = [...this.retryQueue];
      this.retryQueue = [];
      batch.forEach(event => this.queue.push(event));
      this.flush();
    }, 10000);
  }

  startHeartbeat() {
    this.heartbeatInterval = setInterval(async () => {
      try {
        await fetch(HEARTBEAT_URL, {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({ session_id: this.getSessionId() })
        });
      } catch (err) {
        // Fail silently
      }
    }, 30000);
  }

  bindScrollTracking() {
    let trackedLevels = new Set();
    window.addEventListener('scroll', () => {
      const scrollPercent = Math.round((window.scrollY / (document.body.scrollHeight - window.innerHeight)) * 100);
      [25, 50, 75, 100].forEach(level => {
        if (scrollPercent >= level && !trackedLevels.has(level)) {
          trackedLevels.add(level);
          this.track('scroll_depth', { percent: level });
        }
      });
    });
  }

  bindUnloadHandler() {
    window.addEventListener('beforeunload', () => {
      const duration = Math.round((Date.now() - this.sessionStart) / 1000);
      this.track('session_end', { duration });
      // Use beacon for final events
      const finalEvents = [...this.queue];
      finalEvents.forEach(event => {
        navigator.sendBeacon(TRACKING_API_URL, JSON.stringify(event));
      });
    });
  }

  destroy() {
    if (this.flushInterval) clearInterval(this.flushInterval);
    if (this.heartbeatInterval) clearInterval(this.heartbeatInterval);
  }
}

export const tracker = new Tracker();

// Backward compatibility: global trackEvent function
window.trackEvent = (eventType, payload) => {
  tracker.track(eventType, payload);
};
```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Client → /api/v1/track | Untrusted event data from browser |
| Client → /api/v1/heartbeat | Session ID could be spoofed |
| cAdvisor → Host filesystem | Read-only mounts, container escape risk |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-05.1-01 | Tampering | /api/v1/track endpoint | mitigate | Validate TrackingEvent schema with Pydantic, reject malformed payloads |
| T-05.1-02 | Information Disclosure | cAdvisor host mounts | mitigate | All mounts are read-only (:ro), no write access to host filesystem |
| T-05.1-03 | Spoofing | Session ID in tracking events | mitigate | Generate server-side session ID on first visit, validate format |
| T-05.1-04 | Denial of Service | /api/v1/heartbeat flood | mitigate | Rate limit heartbeat to 1 per 30s per session, drop excess requests |
| T-05.1-05 | Elevation of Privilege | cAdvisor container | accept | cAdvisor runs as non-root, read-only mounts, no network exposure beyond localhost |
</threat_model>
